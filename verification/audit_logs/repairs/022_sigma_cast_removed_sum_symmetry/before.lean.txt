import JSP896.CellCounting

/-! The paired-deficit theorem applied to literal residual cell vertices. -/

namespace JSP896.MatchingConfiguration

open FiniteTransversal FiniteGraphCounting
open scoped BigOperators

variable {V κ : Type*} [Fintype V] [Fintype κ] [DecidableEq V] [DecidableEq κ]

/-- After bounded deletions, choose one vertex from every actual cell so that
the only edges among the choices are the prescribed opposite pairs. -/
theorem representatives_after_deletion (G : SimpleGraph V) (P : PairedDeficit.Pairing κ)
    (C R : κ → Finset V) (d : ℕ)
    (hdis : ∀ i j, i ≠ j → Disjoint (C i) (C j))
    (hcomplete : ∀ i, ∀ v ∈ C i, ∀ w ∈ C (P.partner i), G.Adj v w)
    (hdegree : ∀ v, degree G v ≤ d) (hcard : ∀ i, (C i).card ≤ d)
    (hdef : (∑ i, ((d : ℝ) - (C i).card)) < d)
    (hR : ∀ i, R i ⊆ C i) (hremove : (∑ i, (R i).card) ≤ d)
    (hne : ∀ i, (C i \ R i).Nonempty) :
    ∃ f : κ → V, (∀ i, f i ∈ C i \ R i) ∧
      ∀ i j, G.Adj (f i) (f j) ↔ P.partner i = j := by
  classical
  let E (i : κ) := C i \ R i
  let W := (i : κ) × {v // v ∈ E i}
  let value : W → V := fun x => x.2.1
  let index : W → κ := fun x => x.1
  have hvalue (x : W) : value x ∈ C (index x) := (Finset.mem_sdiff.mp x.2.2).1
  have hinj : Function.Injective value := by
    rintro ⟨i, v⟩ ⟨j, w⟩ hvw
    have hij : i = j := by
      by_contra h
      exact Finset.disjoint_left.mp (hdis i j h)
        (Finset.mem_sdiff.mp v.2).1 (hvw.symm ▸ (Finset.mem_sdiff.mp w.2).1)
    subst j
    have hvw' : v = w := Subtype.ext hvw
    subst w
    rfl
  let H : SimpleGraph W := {
    Adj := fun x y => G.Adj (value x) (value y) ∧ index x ≠ index y ∧
      index y ≠ P.partner (index x)
    symm := ⟨by
      intro x y h
      refine ⟨G.adj_symm h.1, h.2.1.symm, ?_⟩
      intro he
      apply h.2.2
      rw [he, P.involutive]⟩
    loopless := ⟨by intro x h; exact h.2.1 rfl⟩ }
  have hsurj (i : κ) : ∃ x : W, index x = i := by
    obtain ⟨v, hv⟩ := hne i
    exact ⟨⟨i, ⟨v, hv⟩⟩, rfl⟩
  have hfiber (i : κ) : (fiber index i).card = (E i).card := by
    apply Finset.card_bij (fun x _ => value x)
    · intro x hx
      have hi : index x = i := (mem_fiber index i x).mp hx
      have hv : value x ∈ E (index x) := x.2.2
      rwa [hi] at hv
    · intro x hx y hy hxy
      exact hinj hxy
    · intro v hv
      refine ⟨⟨i, ⟨v, hv⟩⟩, ?_, rfl⟩
      exact (mem_fiber index i _).mpr rfl
  have hauxDegree (x : W) : degree H x + (C (P.partner (index x))).card ≤ d := by
    let A := (neighbors H x).image value
    have hAcard : A.card = degree H x := Finset.card_image_iff.mpr
      (fun a _ b _ h => hinj h)
    have hAdis : Disjoint A (C (P.partner (index x))) := by
      apply Finset.disjoint_left.mpr
      intro v hv hvC
      obtain ⟨y, hy, rfl⟩ := Finset.mem_image.mp hv
      have hxy := (mem_neighbors H x y).mp hy
      exact Finset.disjoint_left.mp (hdis (index y) (P.partner (index x)) hxy.2.2)
        (hvalue y) hvC
    have hsub : A ∪ C (P.partner (index x)) ⊆ neighbors G (value x) := by
      intro v hv
      apply (mem_neighbors G (value x) v).mpr
      rcases Finset.mem_union.mp hv with hv | hv
      · obtain ⟨y, hy, rfl⟩ := Finset.mem_image.mp hv
        exact ((mem_neighbors H x y).mp hy).1
      · exact hcomplete (index x) (value x) (hvalue x) v hv
    have hc := (Finset.card_le_card hsub).trans (hdegree (value x))
    rw [Finset.card_union_of_disjoint hAdis, hAcard] at hc
    exact hc
  let δ : κ → ℝ := fun i => d - (C i).card
  let ρ : κ → ℝ := fun i => (R i).card
  have hδ (i : κ) : 0 ≤ δ i := by
    dsimp [δ]
    have h : ((C i).card : ℝ) ≤ d := by exact_mod_cast hcard i
    linarith
  have hρ (i : κ) : 0 ≤ ρ i := Nat.cast_nonneg _
  have hremoved : (∑ i, ρ i) ≤ (d : ℝ) := by exact_mod_cast hremove
  have hsize (i : κ) : (d : ℝ) - δ i - ρ i ≤ (fiber index i).card := by
    rw [hfiber]
    have h := Finset.card_sdiff_add_card_eq_card (hR i)
    have hr : ((E i).card : ℝ) + (R i).card = (C i).card := by exact_mod_cast h
    dsimp [δ, ρ]
    linarith
  have hdegreeH (x : W) : (degree H x : ℝ) ≤ δ (P.partner (index x)) := by
    have h : (degree H x : ℝ) + (C (P.partner (index x))).card ≤ d := by
      exact_mod_cast hauxDegree x
    dsimp [δ]
    linarith
  obtain ⟨T, hT, hcolors⟩ := PairedDeficit.hasFull_of_paired_deficit P H index
    (fun _ _ h => h.2.1) (fun _ _ h => h.2.2) hsurj δ ρ d
    hδ hρ hdef hremoved hsize hdegreeH
  have hex (i : κ) : ∃ x ∈ T, index x = i := by
    apply Finset.mem_image.mp
    rw [hcolors]
    exact Finset.mem_univ i
  let f' : κ → W := fun i => Classical.choose (hex i)
  have hf' (i : κ) : f' i ∈ T ∧ index (f' i) = i := Classical.choose_spec (hex i)
  let f : κ → V := fun i => value (f' i)
  have hf (i : κ) : f i ∈ C i \ R i := by
    have h : f i ∈ E (index (f' i)) := (f' i).2.2
    rw [(hf' i).2] at h
    exact h
  refine ⟨f, hf, ?_⟩
  intro i j
  constructor
  · intro hadj
    by_contra hpartner
    have hij : i ≠ j := by
      intro h
      subst j
      exact G.loopless.irrefl (f i) hadj
    apply hT.1 (f' i) (hf' i).1 (f' j) (hf' j).1
    refine ⟨hadj, ?_, ?_⟩
    · simpa only [(hf' i).2, (hf' j).2] using hij
    · simpa only [(hf' i).2, (hf' j).2] using hpartner.symm
  · intro h
    subst j
    exact hcomplete i (f i) (Finset.mem_sdiff.mp (hf i)).1
      (f (P.partner i)) (Finset.mem_sdiff.mp (hf (P.partner i))).1

variable {ι : Type*} [Fintype ι] [DecidableEq ι]

/-- The residual choices themselves form a new full-class IMC. -/
theorem Tree.imc_after_deletion {G : SimpleGraph V} {label : V → ι} {I : Finset V}
    (M : Tree G label I Finset.univ) (hno : ¬ HasFull G label)
    (R : IVertex I → Finset V) (d : ℕ)
    (hdegree : ∀ v, degree G v ≤ d)
    (hdef : (∑ i : IVertex I, ((d : ℝ) - (cell G I i).card)) < d)
    (hR : ∀ i, R i ⊆ cell G I i) (hremove : (∑ i, (R i).card) ≤ d)
    (hne : ∀ i, (cell G I i \ R i).Nonempty) :
    ∃ f : IVertex I → V, (∀ i, f i ∈ cell G I i \ R i) ∧
      Tree G label (Finset.univ.image f) Finset.univ ∧
      ∀ i j, G.Adj (f i) (f j) ↔ M.partner i = j := by
  obtain ⟨f, hf, hadj⟩ := representatives_after_deletion G M.pairing (cell G I) R d
    (fun _ _ h => cells_disjoint G I h)
    (fun i v hv w hw => M.opposite_cells_complete hno i hv hw)
    hdegree (fun i => (cell_card_le_degree G I i).trans (hdegree i.1))
    hdef hR hremove hne
  exact ⟨f, hf, M.exchange_all hno f (fun i => (Finset.mem_sdiff.mp (hf i)).1) hadj, hadj⟩

end JSP896.MatchingConfiguration
